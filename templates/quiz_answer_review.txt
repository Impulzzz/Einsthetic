    }

    function showResults(correctCount) {
        document.getElementById('quiz-container').classList.add('hidden');
        document.getElementById('results-container').classList.remove('hidden');
        
        const incorrectCount = 10 - correctCount;
        const percentage = (correctCount / 10 * 100).toFixed(0);
        
        document.getElementById('score-display').innerText = `${correctCount}/10`;
        document.getElementById('percentage-display').innerText = `${percentage}% Correct`;
        document.getElementById('correct-count').innerText = correctCount;
        document.getElementById('incorrect-count').innerText = incorrectCount;
        
        // Render answer review
        renderAnswerReview();
    }
    
    function renderAnswerReview() {
        const reviewContainer = document.getElementById('answer-review');
        reviewContainer.innerHTML = '';
        
        questions.forEach((q, index) => {
            const userAnswer = userAnswers[index];
            const isCorrect = checkAnswer(userAnswer, q.correct_answer, q.type);
            
            const reviewItem = document.createElement('div');
            reviewItem.className = `border ${isCorrect ? 'border-green-500/30 bg-green-500/5' : 'border-red-500/30 bg-red-500/5'} rounded-xl p-6`;
            
            reviewItem.innerHTML = `
                <div class="flex justify-between items-start mb-4">
                    <div class="flex items-center gap-3">
                        <span class="text-2xl">${isCorrect ? '✅' : '❌'}</span>
                        <span class="font-bold text-lg text-white">Question ${index + 1}</span>
                        <span class="text-xs px-2 py-1 rounded-full ${
                            q.type === 'MCQ' ? 'bg-blue-500/20 text-blue-400' : 
                            q.type === 'MSQ' ? 'bg-purple-500/20 text-purple-400' : 
                            'bg-orange-500/20 text-orange-400'
                        }">${q.type}</span>
                    </div>
                    <span class="text-sm text-gray-400">${isCorrect ? 'Correct' : 'Incorrect'}</span>
                </div>
                
                <div class="mb-4 text-gray-200 leading-relaxed">${q.question}</div>
                
                ${q.options ? `
                    <div class="grid grid-cols-1 md:grid-cols-2 gap-2 mb-4">
                        ${Object.entries(q.options).map(([key, val]) => `
                            <div class="p-3 rounded-lg text-sm ${
                                key === q.correct_answer ? 'bg-green-500/20 border border-green-500/50' :
                                key === userAnswer ? 'bg-red-500/20 border border-red-500/50' :
                                'bg-dark-700/50'
                            }">
                                <span class="font-bold ${
                                    key === q.correct_answer ? 'text-green-400' :
                                    key === userAnswer ? 'text-red-400' :
                                    'text-gray-400'
                                }">${key}.</span> ${val}
                                ${key === q.correct_answer ? ' <span class="text-green-400">✓</span>' : ''}
                                ${key === userAnswer && key !== q.correct_answer ? ' <span class="text-red-400">✗</span>' : ''}
                            </div>
                        `).join('')}
                    </div>
                ` : `
                    <div class="mb-4 space-y-2">
                        <div class="p-3 bg-dark-700 rounded-lg text-sm">
                            <span class="text-gray-400">Your Answer:</span> 
                            <span class="${isCorrect ? 'text-green-400' : 'text-red-400'} font-semibold">${userAnswer || 'No answer'}</span>
                        </div>
                        <div class="p-3 bg-green-500/10 border border-green-500/30 rounded-lg text-sm">
                            <span class="text-gray-400">Correct Answer:</span> 
                            <span class="text-green-400 font-semibold">${q.correct_answer}</span>
                        </div>
                    </div>
                `}
                
                <div class="bg-dark-900 rounded-lg p-4 border border-gray-700">
                    <div class="text-sm font-semibold text-accent-400 mb-2">💡 Explanation:</div>
                    <div class="text-gray-300 text-sm leading-relaxed">${q.explanation}</div>
                </div>
            `;
            
            reviewContainer.appendChild(reviewItem);
        });
        
        // Render MathJax in review
        if (window.MathJax) {
            MathJax.typesetPromise();
        }
    }
